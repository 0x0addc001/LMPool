\begin{frame}{The Placement Mismatch}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{What prefix caching assumes}
        Reuse is cheap when a request executes where its KV prefix already resides.
      \end{block}
      \vspace{0.4em}
      \begin{block}{What data parallelism does}
        Each GPU owns an independent allocator and physical KV cache, while ingress assigns requests globally.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item Round robin may send a matching request to the wrong GPU.
        \item Strict owner routing can overload one replica.
        \item Unconditional KV movement can cost more than recomputation.
      \end{itemize}
      \vspace{0.5em}
      \begin{alertblock}{Design question}
        When should the system route to existing KV, and when should KV move to demand?
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Two Principles}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Routing Principle: Cache Locality}
        Send a request to a GPU that already owns the longest useful contiguous prefix when the saved prefill work exceeds queueing and capacity pressure.
      \end{block}
      \vspace{0.5em}
      \begin{center}
        \Large\good{Route before transfer}
      \end{center}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Transfer Principle: Cache Fluidity}
        Copy or move complete KV block chains over a direct NVLink pair when cache placement no longer matches future demand and expected reuse repays the transaction.
      \end{block}
      \vspace{0.5em}
      \begin{center}
        \Large\good{Transfer only when profitable}
      \end{center}
    \end{column}
  \end{columns}
  \vspace{0.8em}
  \begin{center}
    \large \metric{Avoid recomputation without turning locality into a load hotspot}
  \end{center}
\end{frame}

\begin{frame}{A Request Through \sys{}}
  \small
  \begin{enumerate}
    \item \metric{Ingress:} LLMEngine tokenizes the prompt, builds a Sequence, and computes chained hashes for complete prompt blocks.
    \item \metric{Global route:} Global Scheduler reads versioned prefix, load, and capacity metadata and reserves the selected rank.
    \item \metric{Local admission:} The target Local Scheduler reuses ready local blocks and asks Local Block Manager to allocate missing blocks.
    \item \metric{Optional foreground transfer:} A real block shortage may trigger a cost-gated pair-local copy or move plan.
    \item \metric{Execution:} Model Runner performs prefill, writes missing KV, and then iterates decode.
    \item \metric{Publication:} The worker reports tokens and a new versioned block/load snapshot; only ready blocks become globally visible.
    \item \metric{Completion:} References are released. Complete prefixes remain cached at ref-count zero until reuse or dependency-safe reclamation.
  \end{enumerate}
\end{frame}

\begin{frame}{KV-Aware Routing}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.52\textwidth}
      \small
      For candidate GPU $g$:
      \begin{align*}
        n_g &= \max(0,R-h_g),\\
        M_g &= \min(N,n_gS),\\
        A_g &= \max(0,n_g-F_g),\\
        C_{\mathrm{route}}(g) &= Q_g + w_pM_g + w_eA_gS.
      \end{align*}
      \begin{itemize}
        \item $h_g$: longest contiguous ready prefix
        \item $Q_g$: queued, pending, and running work
        \item $A_g$: dependency-safe reclamation pressure
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{block}{Ingress candidates}
        All healthy replicas; the request has no initial GPU.
      \end{block}
      \begin{block}{Bounded spill}
        Bypass a prefix owner only when its load is sufficiently higher and extra recomputation remains bounded.
      \end{block}
      \begin{alertblock}{Objective}
        Minimize projected completion cost, not maximize hit rate.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Cost-Gated Transfer Admission}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.56\textwidth}
      \small
      \begin{align*}
        T_{\mathrm{data}}^{p}(X) &= \operatorname{PWL}(X;\mathcal{P}_p),\\
        T_{\mathrm{static}}^{p}(X) &= (40\text{ ms}+1.2T_{\mathrm{data}}^{p}(X)),\\
        T_{\mathrm{xfer}} &= \max\{T_{\mathrm{static}},T_{\mathrm{source\ EWMA}},
          T_{\mathrm{placement\ EWMA}}\},\\
        \text{admit} &\Longleftrightarrow T_{\mathrm{save}}\geq\rho T_{\mathrm{xfer}}.
      \end{align*}
    \end{column}
    \begin{column}{0.40\textwidth}
      \begin{itemize}
        \item Pair-specific P95 profile for 1/2/4/8/16/32/64 blocks
        \item Separate nonnegative EWMA residuals by pair and size bucket
        \item Source generation and target capacity revalidated before execution
      \end{itemize}
      \begin{alertblock}{Important qualification}
        40 ms is a conservative manual cold-start prior, not a fitted standalone-pilot result.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Routing Cuts Uncached Prefill}

  \vspace{0.8em}

    \item Uncached prefill falls by \good{50.2\%} and \good{48.6\%}.
    \item Requests remain nearly balanced across ranks.
    \item Latency payoff is model dependent: avoided prefill is more valuable for 1.7B.
  \end{itemize}

      \begin{itemize}
        \item Payload doubles from 4 to 8 blocks.
        \item Latency rises only 52--54\%.
        \item Larger plans amortize launch, gather, synchronization, and scatter.

    This validates and calibrates the data path; it does not by itself prove serving benefit.
  \end{alertblock}

\begin{frame}{Forecast-Assisted Session Handoff}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \begin{block}{Warm-up phase}
        32 session prefixes are created on source GPUs.
      \end{block}
      \vspace{0.5em}
      \begin{block}{Placement boundary}
        The benchmark supplies exact demand for 96 queued reuse requests. \sys{} copies 224 hot blocks in three pair plans.
      \end{block}
      \vspace{0.5em}
      \begin{block}{Reuse phase}
        Placement leases distribute matching requests across source and replica.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{What this experiment proves}
        Routing and transfer can compose when future reuse is known early enough.
      \end{alertblock}
      \begin{alertblock}{What it does not prove}
        The current system does not evaluate an online demand predictor.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Why Routing and Transfer Must Cooperate}
  \small
  \begin{center}
    \begin{tabular}{@{}lrrr@{}}
      \toprule
      Policy & Cached Tokens & Rank Request Ratio & Mean GPU Utilization \\
      \midrule
      Round robin & 47.0\% & 3.3 & 32.2--34.6\% \\
      Transfer only & 70.5\% & 3.3 & 31.0--32.9\% \\
      Full \sys{} & 70.5\% & \good{1.8} & \good{50.2--51.9\%} \\
      \bottomrule
    \end{tabular}
  \end{center}
  \vspace{0.7em}
  \begin{itemize}
    \item Transfer alone removes cold prefill but leaves demand concentrated.
    \item Placement-lease routing turns the copied KV into parallel serving capacity.
    \item Versus routing only, full \sys{} adds 2.6--3.8\% throughput; the 0.6B E2E confidence intervals overlap.
  \end{itemize}
\end{frame}

\begin{frame}{Where Transfer Does Not Help}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Memory skew}
        Qwen3-1.7B admits no transfer. Cached-token ratios remain near 55\%, and full-system throughput is 0.8\% below round robin.
      \end{block}
      \vspace{0.4em}
      \small Pressure alone is insufficient without reusable state, capacity, and future demand.
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Steady load skew}
        Round robin naturally warms the same hot prefix on all six workers. Every multi-GPU throughput result remains within 0.6\% of baseline.
      \end{block}
      \vspace{0.4em}
      \small Once locality is already replicated, movement cannot remove additional work.
    \end{column}
  \end{columns}
  \vspace{0.8em}
  \begin{center}
    \large\caution{A useful KV pool rejects unprofitable movement.}
  \end{center}
\end{frame}

\begin{frame}{Limitations and Future Work}
  \small
  \begin{itemize}
    \item Synthetic traces; exact future-demand signal in the positive handoff workload
    \item Internal Mini-vLLM ablations rather than production-engine comparisons
    \item Six RTX 3090 GPUs, two sub-2B models with identical KV geometry
    \item Fixed 256-token block size; no block-size sensitivity study
    \item Manual 40 ms transaction prior; no held-out cost-prediction error or policy ablation
    \item Single control process and launcher; no consensus or launcher failover
  \end{itemize}
  \vspace{0.5em}
  \begin{block}{Next evaluation}
    Online forecast on public multi-turn traces; production baseline; 1/2/4/6-GPU and block-size scaling; cost-model prediction and admission ablations.
  \end{block}
\end{frame}

\begin{frame}{Takeaways}
  \begin{enumerate}
    \item \metric{Routing solves cache locality:} reuse KV where it already resides without creating a load hotspot.
    \item \metric{NVLink transfer provides cache fluidity:} move complete block chains only when reuse repays the full transaction.
    \item \metric{The mechanisms must cooperate:} copied KV matters only when routing sends demand to the replica.
    \item \metric{The boundary is part of the result:} memory or request skew does not automatically justify transfer.
  \end{enumerate}
  \vspace{1em}
  \begin{center}
    \Large\good{Route when possible. Transfer when profitable.}
  \end{center}
